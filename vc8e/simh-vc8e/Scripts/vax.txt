;
; Load CPU microcode
load -r ka655.bin
;
; Attach non-volatile RAM to a file
attach nvr nvram.bin
;
; This virtual machine has 64M memory
set cpu 64m
;
; Define disk drive types. RA92 is largest-supported VAX drive.
set rq0 ra92
set rq1 ra92
set rq2 ra92
set rq3 cdrom
;
; Attach defined drives to local files
attach rq0 ../../vms/d0.dsk
attach rq1 ../../vms/d1.dsk
attach rq2 ../../vms/d2.dsk
;
; Attach the CD-ROM to its file (read-only) 
; attach -r rq3 OpenVMS-VAX-7.3.iso
;
; Disable unused devices. It's also possible to disable individual devices,
; using a construction like "set rq2 disable" if desired. 
;
set rl disable
set ts disable
set tq disable
set qvss cap
set qvss enable
; 
; Attach Ethernet to a network interface
set xq mac=08-00-2B-AA-BB-CC
attach xq eth3
;
; Now start the emulator
boot cpu
